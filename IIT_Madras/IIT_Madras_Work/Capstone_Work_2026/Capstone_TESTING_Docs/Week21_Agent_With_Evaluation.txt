##STEP    8: TOOLS

def knowledge_tool_func(query):
docs = vector_store.similarity_search(query, k=3)
if len(docs) == 0:
return “No relevant information found”
return “\n”.join([d.page_content for d in docs])

def escalation_tool_func(query):
return “Escalation triggered for: “ + query

knowledge_tool = Tool(
name=“KnowledgeBaseSearch”,
func=knowledge_tool_func,
description=“Search KB”
)

escalation_tool = Tool(
name=“EscalationTool”,
func=escalation_tool_func,
description=“Escalate”
)

tools = [knowledge_tool, escalation_tool]

support_crew = Crew(

failure_analysis_agent
],
tasks=[
intent_task,
retrieval_task,
reasoning_task,

failure_task
],

)

##STEP    14: EXECUTION of Crew FUNCTION WITH EVALUATION logic and approach 

def run_with_evaluation(query):
start = time.time()

try:
result = support_crew.kickoff(inputs={"query": query})

latency = time.time() - start

logger.info("SUCCESS")
logger.info("Latency: " + str(latency))

return {
"result": str(result),
"latency": latency,
"status": "success"
}

except Exception:
logger.error(traceback.format_exc())
return {
"result": "Error occurred. Escalated.",
"status": "failed"
}
