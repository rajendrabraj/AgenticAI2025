##STEP    8: TOOLS

def knowledge_tool_func(query):
docs = vector_store.similarity_search(query, k=3)
if len(docs) == 0:
return “No relevant information found”
return “\n”.join([d.page_content for d in docs])

def escalation_tool_func(query):
return “Escalation triggered for: “ + query

knowledge_tool = Tool(
name=“KnowledgeBaseSearch”,
func=knowledge_tool_func,
description=“Search KB using embeddings”
)

escalation_tool = Tool(
name=“EscalationTool”,
func=escalation_tool_func,
description=“Escalate to human agent”
)

tools = [knowledge_tool, escalation_tool]

support_crew = Crew(

intent_agent,

reasoning_agent,
escalation_agent
],
tasks=[
intent_task,
memory_task,
retrieval_task,
reasoning_task,
escalation_task
],
verbose=False
)


##=====================================================================================

##STEP    13: EXECUTION WITH LOGGING

def execute_with_tracking(query):
start_time = time.time()

try:
result = support_crew.kickoff(inputs={
"query": query,
"memory": str(short_term_memory.load_memory_variables({}))
})

latency = time.time() - start_time

logger.info("SUCCESS")
logger.info("Latency: " + str(latency))

return {
"result": str(result),
"latency": latency,
"status": "success"
}

except Exception:
latency = time.time() - start_time

logger.error("FAILURE")
logger.error(traceback.format_exc())

return {
"result": "System error. Escalating to human agent.",
"latency": latency,
"status": "failed"
}

##STEP    14: API ENDPOINT

@app.post(”/support”)
def support_endpoint(request: dict):
query = request.get(“query”)

if not query:
raise HTTPException(status_code=400, detail="Query missing")

return execute_with_tracking(query)
