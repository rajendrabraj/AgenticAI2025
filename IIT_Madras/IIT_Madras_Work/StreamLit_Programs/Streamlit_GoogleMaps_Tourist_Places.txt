def get_places(location, radius, keyword):
    """
    Uses the Google Maps Places API to find places near a given location.

    Args:
        location (str): The location to search near, e.g., "Pune, Maharashtra".
        radius (int): The radius in meters to search within.
        keyword (str): Keywords to filter the search results.

    Returns:
        pandas.DataFrame: A DataFrame containing the search results, or None if an error occurred.
    """
    try:
        gmaps = googlemaps.Client(key=google_cloud_apikey)
        places_result = gmaps.places(query=keyword, location=location, radius=radius)

        if places_result and places_result['results']:
            places_data = []
            for place in places_result['results']:
                name = place.get('name')
                address = place.get('formatted_address')
                latitude = place['geometry']['location']['lat']
                longitude = place['geometry']['location']['lng']
                place_id = place.get('place_id')
                places_data.append({
                    'Name': name,
                    'Address': address,
                    'Latitude': latitude,
                    'Longitude': longitude,
                    'Place ID': place_id
                })
            return pd.DataFrame(places_data)
        else:
            st.error("No places found for the given criteria.")
            return None
    except googlemaps.exceptions.ApiError as e:
        st.error(f"Google Maps API error: {e}")
        return None
    except Exception as e:
        st.error(f"An error occurred: {e}")
        return None

def main():
    """
    Main function to run the Streamlit app.
    """
    st.title("Google Maps Place Finder")

    location = st.text_input("Enter a location (e.g., Pune, Maharashtra):", "Pune, Maharashtra")
    radius = st.number_input("Enter the search radius in meters:", min_value=100, max_value=50000, value=5000)
    keyword = st.text_input("Enter keywords (e.g., restaurant, cafe):", "restaurant")

    if st.button("Find Places"):
        if google_cloud_apikey is None:
            st.error("Please set the google_cloud_apikey environment variable.")
            return

        if not location or not keyword:
            st.warning("Please enter both a location and keywords.")
            return
        
        df_places = get_places(location, radius, keyword)

        if df_places is not None:
            st.subheader("Search Results:")
            st.dataframe(df_places)

            st.subheader("Map:")
            st.map(df_places, latitude='Latitude', longitude='Longitude')


if __name__ == "__main__":
    main()
