import streamlit as st
import pandas as pd
from io import StringIO

st.title("Display Comma-Separated Data as HTML Table")

# Sample comma-separated records (as a single string)
csv_data = """
Name,Age,City
Alice,25,New York
Bob,30,San Francisco
Charlie,35,Los Angeles
"""

# Optional: show the raw input
with st.expander("Raw CSV data"):
st.text(csv_data.strip())

# Convert string to a DataFrame
df = pd.read_csv(StringIO(csv_data))

# Show DataFrame (optional)
st.subheader("DataFrame Preview")
st.dataframe(df)

# Convert DataFrame to HTML table
html_table = df.to_html(index=False, border=1)

# Show raw HTML (optional)
with st.expander("See raw HTML code"):
st.code(html_table, language='html')

# Render the HTML table
st.subheader("HTML Table Output")
st.markdown(html_table, unsafe_allow_html=True)
